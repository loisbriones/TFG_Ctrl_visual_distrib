\chapter{Códigos de referencia}
\label{chap:listados}

\lettrine{E}{n} este anexo se recogen íntegras las definiciones de los mensajes propios del sistema (sección~\ref{subsec:mensajes-propios}) y los ficheros que definen su configuración y despliegue (sección~\ref{sec:configuracion_despliegue}), junto con la política que se emplea en la prueba de la sección~\ref{sec:prueba-politica-optima}. El único que aparece en el cuerpo de la memoria es el mensaje \texttt{CarLocation} (código~\ref{lst:msg-carlocation}), que sirve de ejemplo del formato de los demás. Todos los ficheros se muestran sin sus comentarios, que pueden consultarse en el repositorio del proyecto~\cite{repogithub}.

def generate_launch_description():

    params_file = PathJoinSubstitution([
        FindPackageShare('image_processor_pkg'),
        'config',
        'params.yaml'
    ])

    node_id = DeclareLaunchArgument(
        'node_id',
        default_value='camera_00',
        description='Nombre con el que se va a identificar al nodo dentro de la red de ROS2'

    )

    camera_name = DeclareLaunchArgument(
        'camera_name',
        default_value='0',
        description='ID numerico que tiene la ruta (0) o ruta entera de la camara (/dev/video0)'
    )

    return LaunchDescription([
        camera_name,
        node_id,
        Node(
            package='image_processor_pkg',
            executable='CameraNode.py',
            namespace=LaunchConfiguration('node_id'),
            parameters=[
                params_file,
                {'camara_id': LaunchConfiguration('node_id')},
                {'camera.device': LaunchConfiguration('camera_name')}
            ],
            respawn=True,
            respawn_delay=2.0
        )
    ])

\begin{lstlisting}[style=codigo, language=Python, caption={Lanzamiento de los controladores y del nodo de potencia (\texttt{launch/ControlLaunch.py})}, label={lst:control-launch}]
def generate_launch_description():
    pkg_share = get_package_share_directory("image_processor_pkg")
    params_file = os.path.join(pkg_share, "config", "params.yaml")

    with open(params_file, "r") as f:
        config = yaml.safe_load(f)
        params = config["/**"]["ros__parameters"]
        coches = params["coches"]
        cars = params["cars"]

    ld = LaunchDescription()

    hay_autonomo = False
    for car_name in coches:
        cfg = cars[car_name]
        modo = cfg.get("modo", "automatico")
        if modo == "manual":
            continue
        hay_autonomo = True

        ld.add_action(Node(
            package="image_processor_pkg", executable="CarControllerNode.py",
            name=f"CarController_{modo}_{car_name}", namespace=car_name,
            parameters=[params_file, {"car_name": car_name,
                                      "carril_asignado": str(cfg["carril"]),
                                      "modo": modo}],
            respawn=True, respawn_delay=2.0))

    if hay_autonomo:
        ld.add_action(Node(
            package="image_processor_pkg", executable="RaceControllerNode.py",
            name="arduino_bridge", parameters=[params_file]))

    if params.get("net_probe", {}).get("enabled", True):
        ld.add_action(Node(
            package="image_processor_pkg", executable="NetProbeNode.py",
            name="net_probe", parameters=[params_file],
            respawn=True, respawn_delay=2.0))

    return ld
\end{lstlisting}

\begin{lstlisting}[style=codigo, language=Python, caption={Lanzamiento en modo manual, sin nodo de potencia (\texttt{launch/ManualLaunch.py})}, label={lst:manual-launch}]
def generate_launch_description():
    pkg_share = get_package_share_directory("image_processor_pkg")
    params_file = os.path.join(pkg_share, "config", "params.yaml")

    with open(params_file, "r") as f:
        config = yaml.safe_load(f)
        params = config["/**"]["ros__parameters"]
        coches = params["coches"]
        cars = params["cars"]

    ld = LaunchDescription()

    for car_name in coches:
        cfg = cars[car_name]
        if cfg.get("modo", "automatico") != "manual":
            continue

        ld.add_action(
            Node(
                package="image_processor_pkg",
                executable="CarControllerNode.py",
                name=f"CarControllerManual_{car_name}",
                namespace=car_name,
                parameters=[
                    params_file,
                    {
                        "car_name": car_name,
                        "carril_asignado": str(cfg["carril"]),
                        "modo": "manual",
                    },
                ],
                respawn=True,
                respawn_delay=2.0
            )
        )

    return ld
\end{lstlisting}

\section{Ficheros de despliegue en contenedores}

\begin{lstlisting}[style=codigo, language=yamlsimple, caption={Imagen común de todos los nodos (\texttt{Dockerfile})}, label={lst:dockerfile}]
FROM ros:humble-ros-base

ENV DEBIAN_FRONTEND=noninteractive

RUN apt-get update && apt-get install -y \
  ros-humble-rosidl-default-generators \
  ros-humble-rosidl-default-runtime \
  python3-opencv \
  python3-numpy \
  python3-serial \
  tzdata \
  && rm -rf /var/lib/apt/lists/*

WORKDIR /ros2_ws

RUN echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc
\end{lstlisting}
